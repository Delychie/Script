--!nonstrict

-- LinearMovement: moves you with a LinearVelocity constraint instead of raw velocity writes.
-- Native humanoid movement still handles animation, jumping, rooting and the game's own
-- WalkSpeed (e.g. carry slow-downs). Two modes:
--   LINEAR_PRIMARY = false (default) -- booster. The constraint only engages when your effective
--     speed is above the game's, and drives the extra speed. Otherwise movement is pure vanilla.
--   LINEAR_PRIMARY = true            -- the constraint drives ALL walking, at whatever speed the
--     game allows (with SPEED_MULT = 1 that is exactly the game's WalkSpeed).
-- Primary mode changes the MECHANISM, not the speed. Moving faster than the game allows is what
-- gets noticed, and that comes from SPEED_MULT / SPEED_FIXED -- not from which mode you pick.

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")

local player = Players.LocalPlayer

-- Speed: effective = SPEED_FIXED (if > 0) else the game's current WalkSpeed * SPEED_MULT.
-- Multiply mode respects carry slow-downs / boosts and never fights the game's rooting.
local SPEED_MULT = 1        -- multiplier on the game's live WalkSpeed (default 1 = follow the game's speed exactly)
local SPEED_FIXED = 0       -- 0 = not fixed; speed follows the game (WalkSpeed * SPEED_MULT). Set > 0 only to hard-lock a speed.

-- LINEAR_PRIMARY: drive ALL walking through the constraint, not only the part above the game's speed.
-- With SPEED_MULT = 1 this moves you at exactly the game's WalkSpeed -- same speed as vanilla, but the
-- plane constraint is what moves you. Native movement still owns animation, jumping and rooting.
-- NOTE: this changes the MECHANISM, not the speed. It is not by itself safer or less safe; going
-- faster than the game allows is what gets noticed, and that is SPEED_MULT / SPEED_FIXED, not this.
local LINEAR_PRIMARY = false
local AIR_CONTROL = true    -- steer while airborne (only matters while the constraint is driving)
-- CARRY_SAFE: whenever the game slows you below your normal speed -- carrying something, a debuff,
-- any deliberate slow-down -- stand fully down. No boost, no linearizing, no constraints: pure
-- vanilla movement. The game slowing you is the game telling you how fast you may go, and moving
-- faster than that while carrying is precisely what a carry check looks for. Leave this ON.
local CARRY_SAFE = true
local CARRY_SLOW_MARGIN = 0.5 -- studs/s below your observed normal speed that counts as "slowed"
-- Dominance: force other scripts' movement off so only this one moves you.
-- SUPPRESS_OTHER wipes foreign BodyVelocity/LinearVelocity/etc off your HRP each frame.
-- MASK_WALKSPEED (>0) pins WalkSpeed to that value so other scripts' WalkSpeed spikes are hidden;
-- pair it with SPEED_FIXED for your real speed. NOTE: neither can hide a raw AssemblyLinearVelocity
-- write or beat a server displacement check - for those, disable the other script's mover instead.
local SUPPRESS_OTHER = false
local MASK_WALKSPEED = 0    -- 0 = don't touch WalkSpeed (leave default). WARNING: a value < ~25 makes carry detectors think you're always carrying.
-- Executor-only: no-op ANY Lua write to Velocity / AssemblyLinearVelocity on your HRP, so other
-- scripts (e.g. del hub) that move you by writing velocity are neutralised and only this constraint
-- moves you. Needs hookmetamethod + newcclosure. Read the caveats before enabling it.
local BLOCK_FOREIGN_VELOCITY = false
-- Executor-only: while you're RAGDOLLED / knocked (hit by a bat, flung, getting up), no-op any other
-- script's velocity writes on your HRP so the ragdoll plays out naturally and you don't get kicked for
-- "moving while limp". Scoped strictly to the knocked window, so it never touches walking/jumps/auto-walk.
-- Does NOT stop the hit or the ragdoll itself. Needs hookmetamethod + newcclosure.
local SUPPRESS_DURING_RAGDOLL = true
local SLOPE_GLUE = true     -- hug slopes instead of launching off them at high speed
local STEP_ASSIST = true    -- glide up small ledges/stairs while boosting
local STEP_HEIGHT = 3       -- tallest ledge treated as a step (studs)
local STEP_PROBE = 1.8      -- base look-ahead for a step (studs; grows with speed)
local STEP_UP_SPEED = 40    -- base rise rate over a step (studs/s; grows with speed)
local GROUND_TOLERANCE = 0.9
local FLOOR_VEL_DEADZONE = 0.25

-- LINEARIZE_FOREIGN (executor-only): a general "make it linear" transformer. Instead of hard-writing
-- to your HRP, other scripts' movement is CAUGHT and reproduced through this script's LinearVelocity
-- constraint. It doesn't know or care which script issued the move -- anything that writes HRP velocity
-- or CFrame gets turned linear while this is on:
--   * a raw velocity write (e.g. an anti-ragdoll's Velocity = 0)  -> reproduced as a linear velocity
--   * a CFrame teleport (e.g. an auto-TP snapping you to the floor) -> reproduced as a fast linear glide
-- So you don't rewrite those scripts; you point this at them and their motion comes out linear.
-- Needs hookmetamethod + newcclosure.
-- WARNING: while ON, LINEARIZE_TELEPORTS catches EVERY local HRP CFrame/Position write, including any
-- legit ones, and turns them into glides. Leave teleports off if you only want velocity linearized.
-- Caught motion is classified by INTENT, because the same write means different things:
--   Set      (Velocity = v, v ~= 0) -> drive v briefly, then release to physics
--   Damp     (Velocity = 0)         -> ONE-SHOT BRAKE that bleeds momentum, never a sustained hold
--   Impulse  (a knock/launch)       -> reproduce with a decay curve so it settles naturally
--   Teleport (CFrame/Position =)    -> glide to the target, easing in
-- Damp is why zeroing writes don't freeze you: an anti-ragdoll that writes Velocity = 0 every frame
-- is asking you to stop, not asking to be pinned in mid-air.
local LINEARIZE_FOREIGN = false  -- master toggle for the transformer
local LINEARIZE_TELEPORTS = true -- also catch CFrame/Position teleports, not just velocity writes
local LINEARIZE_RAGDOLL = true   -- linearize while knocked too, instead of dropping the writes
local LINEARIZE_ANGULAR = true   -- reproduce spin/tumble (RotVelocity) as well as travel
local LINEARIZE_GRAVITY = true   -- leave the Y axis to gravity unless an intent really is vertical
local LINEARIZE_SPEED = 200      -- max studs/s used when gliding to a caught teleport target
local LINEARIZE_REACH = 1.5      -- a glide target counts as reached within this many studs
local LINEARIZE_TTL = 0.15       -- how long a caught Set keeps driving (seconds)
local LINEARIZE_DAMP_TIME = 0.12 -- how long a Damp takes to bleed momentum to zero (seconds)
local LINEARIZE_DECAY = 2.5      -- decay rate for Impulse intents (higher = settles sooner)
local LINEARIZE_FORCE = 50000    -- finite force; math.huge snaps hard and reads as teleport-y
-- Coverage: a velocity write only has to land on SOME part of your assembly to move all of you, and
-- plenty of scripts move you by method call or by parenting a mover instead of writing a property.
local LINEARIZE_WHOLE_RIG = true -- catch writes on any part of your character, not just the HRP
local LINEARIZE_METHODS = true   -- catch :ApplyImpulse / :PivotTo / :MoveTo / :TranslateBy (__namecall)
local LINEARIZE_MOVERS = true    -- catch foreign BodyVelocity/VectorForce/etc parented to your rig
-- Diagnostics: log every caught channel WITHOUT converting anything, so you can see what a given
-- game/script actually does to you (and how it ragdolls) before you turn conversion on.
local LINEARIZE_DEBUG = false
-- Leak detector: each frame, compare your ACTUAL velocity against what our constraints commanded and
-- report any unexplained jump. This is how you find velocity arriving through a channel we cannot
-- hook at all (the server, replication, or a script that hooked the metamethod above us).
local LINEARIZE_AUDIT = false
local AUDIT_TOLERANCE = 12       -- studs/s of unexplained change before it gets reported
local AUDIT_COOLDOWN = 0.5       -- min seconds between reports, so the log stays readable
-- Logging. The property hook runs on EVERY write in the game, so logging is throttled per channel --
-- unthrottled printing is enough to stall the client on its own. Repeats are counted, not printed.
local LOG_THROTTLE = 0.75        -- min seconds between log lines for the same channel
-- Persist the log to a file so it survives losing the session. Without this, anything logged in the
-- last moments before a disconnect is gone with the output window, which is exactly the window you
-- care about when something ends your session abruptly. Needs an executor with writefile/appendfile.
local LOG_TO_FILE = false
local LOG_FILE = "LinearMovement_log.txt"
local LOG_FLUSH_EVERY = 0.5      -- seconds between file writes (batched; never write per-event)

local humanoid: Humanoid? = nil
local rootPart: BasePart? = nil
local attachment: Attachment? = nil
local moveLV: LinearVelocity? = nil
local stepLV: LinearVelocity? = nil
local linLV: LinearVelocity? = nil     -- horizontal (XZ plane) reproduction
local linYLV: LinearVelocity? = nil    -- vertical (Y line), only when an intent is truly vertical
local linAV: AngularVelocity? = nil    -- spin/tumble reproduction
local rayParams: RaycastParams? = nil

local footOffset = 3
local stepping = false
local stepTargetY = 0
local stepDeadline = 0
local jumpSuppressUntil = 0
local knockUntil = 0
local lastGlueTime = 0
local lastGlueVy = 0
local airFloorVX, airFloorVZ = 0, 0
local baseWalkSpeed = 0   -- the fastest WalkSpeed this life; anything under it means the game slowed you
-- transformer state: the latest caught foreign motion, normalized into one intent record
local linKind: string? = nil         -- "Set" | "Damp" | "Impulse" | "Teleport"
local linVec = Vector3.zero          -- the velocity that intent wants (or the momentum being bled)
local linUntil = 0                   -- when this intent expires and we hand back to physics
local linStart = 0                   -- when it began (used to shape the Damp/Impulse curve)
local linPosTarget: Vector3? = nil   -- a caught teleport target to glide to
local linNoY = false                 -- intent only meant to change horizontal motion; leave Y alone
local linAngVec = Vector3.zero       -- caught angular velocity to reproduce
local linAngUntil = 0
local charGen = 0

-- Every BasePart of your character, kept as a set so the hooks can test membership in O(1). A hook
-- runs on EVERY property write in the game, so it must never walk the tree to decide.
local ourParts: { [Instance]: boolean } = {}
-- leak-detector state
local auditLastVel = Vector3.zero
local auditLastCmd = Vector3.zero
local auditLastReport = 0
-- logging state: throttle bookkeeping plus a buffer that is batched out to disk
local logLast: { [string]: number } = {}
local logSkipped: { [string]: number } = {}
local logBuf: { string } = {}
local logNextFlush = 0

-- forward declarations: these are defined further down but referenced from onCharacterAdded above them
local noteIntent, noteAngular, logAssemblies, linLog

local KNOCK_THRESHOLD = 50   -- horizontal speed over expected that counts as external knockback

local connections: { RBXScriptConnection } = {}
local alive = true

local function teardown()
	if moveLV then moveLV:Destroy() moveLV = nil end
	if stepLV then stepLV:Destroy() stepLV = nil end
	if linLV then linLV:Destroy() linLV = nil end
	if linYLV then linYLV:Destroy() linYLV = nil end
	if linAV then linAV:Destroy() linAV = nil end
	if attachment then attachment:Destroy() attachment = nil end
end

if _G.__LinearMovementCleanup then
	pcall(_G.__LinearMovementCleanup)
end

local function purgeOld(part: BasePart)
	for _, c in ipairs(part:GetChildren()) do
		local n = c.Name
		if n == "LinearMoveAttachment" or n == "LinearMove" or n == "LinearJump" or n == "LinearStep"
			or n == "LinearForce" or n == "LinearForceY" or n == "LinearSpin" then
			c:Destroy()
		end
	end
end

local function build()
	teardown()
	if not rootPart then
		return
	end
	purgeOld(rootPart)

	local att = Instance.new("Attachment")
	att.Name = "LinearMoveAttachment"
	att.Parent = rootPart

	local move = Instance.new("LinearVelocity")
	move.Name = "LinearMove"
	move.Attachment0 = att
	move.RelativeTo = Enum.ActuatorRelativeTo.World
	move.VelocityConstraintMode = Enum.VelocityConstraintMode.Plane
	move.PrimaryTangentAxis = Vector3.new(1, 0, 0)
	move.SecondaryTangentAxis = Vector3.new(0, 0, 1)
	move.MaxForce = math.huge
	move.PlaneVelocity = Vector2.zero
	move.Enabled = false
	move.Parent = rootPart

	local step = Instance.new("LinearVelocity")
	step.Name = "LinearStep"
	step.Attachment0 = att
	step.RelativeTo = Enum.ActuatorRelativeTo.World
	step.VelocityConstraintMode = Enum.VelocityConstraintMode.Line
	step.LineDirection = Vector3.new(0, 1, 0)
	step.MaxForce = math.huge
	step.LineVelocity = 0
	step.Enabled = false
	step.Parent = rootPart

	-- Linearizer, split by axis so gravity survives: horizontal goes through a Plane (XZ) constraint
	-- and vertical through a separate Line (Y) constraint that only switches on for genuinely
	-- vertical intents. A single Vector-mode constraint would drive all three axes and make you float.
	local lin = Instance.new("LinearVelocity")
	lin.Name = "LinearForce"
	lin.Attachment0 = att
	lin.RelativeTo = Enum.ActuatorRelativeTo.World
	lin.VelocityConstraintMode = Enum.VelocityConstraintMode.Plane
	lin.PrimaryTangentAxis = Vector3.new(1, 0, 0)
	lin.SecondaryTangentAxis = Vector3.new(0, 0, 1)
	lin.MaxForce = LINEARIZE_FORCE
	lin.PlaneVelocity = Vector2.zero
	lin.Enabled = false
	lin.Parent = rootPart

	local linY = Instance.new("LinearVelocity")
	linY.Name = "LinearForceY"
	linY.Attachment0 = att
	linY.RelativeTo = Enum.ActuatorRelativeTo.World
	linY.VelocityConstraintMode = Enum.VelocityConstraintMode.Line
	linY.LineDirection = Vector3.new(0, 1, 0)
	linY.MaxForce = LINEARIZE_FORCE
	linY.LineVelocity = 0
	linY.Enabled = false
	linY.Parent = rootPart

	-- ragdolls tumble; without this a knock travels linearly but never rotates, which looks wrong
	local spin = Instance.new("AngularVelocity")
	spin.Name = "LinearSpin"
	spin.Attachment0 = att
	spin.RelativeTo = Enum.ActuatorRelativeTo.World
	spin.AngularVelocity = Vector3.zero
	spin.MaxTorque = LINEARIZE_FORCE
	spin.Enabled = false
	spin.Parent = rootPart

	attachment = att
	moveLV = move
	stepLV = step
	linLV = lin
	linYLV = linY
	linAV = spin
end

local function computeFootOffset(char: Model, hum: Humanoid, hrp: BasePart): number
	local off = hrp.Size.Y / 2 + hum.HipHeight
	if hum.RigType == Enum.HumanoidRigType.R6 then
		local leg = char:FindFirstChild("Left Leg") or char:FindFirstChild("Right Leg")
		if leg and leg:IsA("BasePart") then
			off += leg.Size.Y
		else
			off = math.max(off, 3)
		end
	end
	return off
end

-- Names of instance types that move you without ever writing a velocity property.
local MOVER_CLASSES = {
	BodyVelocity = true, BodyForce = true, BodyThrust = true, BodyPosition = true,
	BodyGyro = true, BodyAngularVelocity = true, RocketPropulsion = true,
	LinearVelocity = true, AngularVelocity = true, VectorForce = true,
	AlignPosition = true, AlignOrientation = true, Torque = true,
}

-- A foreign mover appeared on the rig. Neutralize it by zeroing its force rather than destroying it
-- (its owning script keeps its reference and won't error or respawn it in a loop), then reproduce
-- its intent through our constraints so the motion still happens -- just linearly.
local function absorbMover(inst: Instance)
	if not MOVER_CLASSES[inst.ClassName] then
		return
	end
	local n = inst.Name
	if n == "LinearMove" or n == "LinearStep" or n == "LinearForce"
		or n == "LinearForceY" or n == "LinearSpin" then
		return -- ours
	end
	linLog("mover", ("%s %q parented to %s"):format(inst.ClassName, n, tostring(inst.Parent)))
	if not LINEARIZE_FOREIGN or not LINEARIZE_MOVERS or LINEARIZE_DEBUG then
		return
	end
	pcall(function()
		if inst:IsA("BodyVelocity") then
			noteIntent("Set", inst.Velocity)
			inst.MaxForce = Vector3.zero
		elseif inst:IsA("LinearVelocity") then
			if inst.VelocityConstraintMode == Enum.VelocityConstraintMode.Vector then
				noteIntent("Set", inst.VectorVelocity)
			end
			inst.MaxForce = 0
		elseif inst:IsA("BodyAngularVelocity") then
			noteAngular(inst.AngularVelocity)
			inst.MaxTorque = Vector3.zero
		elseif inst:IsA("AngularVelocity") then
			noteAngular(inst.AngularVelocity)
			inst.MaxTorque = 0
		elseif inst:IsA("BodyPosition") then
			linPosTarget = inst.Position
			inst.MaxForce = Vector3.zero
		elseif inst:IsA("AlignPosition") then
			inst.MaxForce = 0
		elseif inst:IsA("VectorForce") or inst:IsA("BodyForce") or inst:IsA("BodyThrust") then
			-- a force is an acceleration, not a velocity; just neutralize it
			if inst:IsA("VectorForce") then inst.Force = Vector3.zero else inst.Force = Vector3.zero end
		end
	end)
end

local function trackParts(char: Model)
	table.clear(ourParts)
	for _, d in ipairs(char:GetDescendants()) do
		if d:IsA("BasePart") then
			ourParts[d] = true
		end
	end
end

local function onCharacterAdded(char: Model)
	charGen += 1
	local myGen = charGen
	local hum = char:WaitForChild("Humanoid") :: Humanoid
	if not alive or myGen ~= charGen or not hum then
		return
	end
	local hrp = char:WaitForChild("HumanoidRootPart") :: BasePart
	if not alive or myGen ~= charGen or not hrp then
		return
	end
	humanoid = hum
	rootPart = hrp
	-- populate the part set BEFORE anything else can write to us, so respawns don't leak a window
	-- where the hooks don't recognise the new rig yet
	trackParts(char)
	table.insert(connections, char.DescendantAdded:Connect(function(d)
		if d:IsA("BasePart") then
			ourParts[d] = true
		else
			absorbMover(d)
		end
	end))
	table.insert(connections, char.DescendantRemoving:Connect(function(d)
		ourParts[d] = nil
	end))
	footOffset = computeFootOffset(char, hum, hrp)
	build()

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { char }
	params.IgnoreWater = true
	params.RespectCanCollide = true
	rayParams = params

	stepping = false
	linKind = nil
	linPosTarget = nil
	linNoY = false
	linVec = Vector3.zero
	linAngVec = Vector3.zero
	linAngUntil = 0
	jumpSuppressUntil = 0
	knockUntil = 0
	lastGlueTime = 0
	airFloorVX, airFloorVZ = 0, 0
	baseWalkSpeed = hum.WalkSpeed

	-- also suppress step-assist on programmatic jumps (not just JumpRequest input)
	table.insert(connections, hum.StateChanged:Connect(function(_, new)
		if new == Enum.HumanoidStateType.Jumping then
			jumpSuppressUntil = tick() + 0.25
			stopStep()
		end
	end))

	-- State-change trace. Humanoid state REPLICATES to the server, so this log is a record of what
	-- the server is being told about you. A knocked state followed immediately by a forced Running
	-- (see the "CONTRADICTS" line) is another script cancelling a ragdoll the server applied -- the
	-- client asserting it is not ragdolled while the server knows it is.
	local lastKnockAt = 0
	table.insert(connections, hum.StateChanged:Connect(function(old, new)
		local now = tick()
		local knocked = new == Enum.HumanoidStateType.Ragdoll
			or new == Enum.HumanoidStateType.Physics
			or new == Enum.HumanoidStateType.FallingDown
			or new == Enum.HumanoidStateType.PlatformStanding
		if knocked then
			lastKnockAt = now
		end
		linLog("state", ("%s -> %s"):format(tostring(old), tostring(new)))
		if new == Enum.HumanoidStateType.Running and lastKnockAt > 0 and now - lastKnockAt < 0.5 then
			linLog("state", ("CONTRADICTS: forced Running %.2fs after a knock -- something is cancelling"
				.. " the ragdoll, and that disagreement replicates to the server"):format(now - lastKnockAt))
		end
	end))

	-- Knock takeover: the moment you get ragdolled, sample the momentum the hit gave you and hand it
	-- to the constraint as an Impulse. From frame one the flight arc is constraint-driven rather than
	-- raw physics, which is the whole point -- getting knocked should come out linear too.
	table.insert(connections, hum.StateChanged:Connect(function(_, new)
		local knocked = new == Enum.HumanoidStateType.Ragdoll
			or new == Enum.HumanoidStateType.Physics
			or new == Enum.HumanoidStateType.FallingDown
			or new == Enum.HumanoidStateType.PlatformStanding
		if not knocked then
			return
		end
		logAssemblies(tostring(new))
		if not LINEARIZE_FOREIGN or not LINEARIZE_RAGDOLL or LINEARIZE_DEBUG then
			return
		end
		local hrpNow = rootPart
		if not hrpNow then
			return
		end
		noteIntent("Impulse", hrpNow.AssemblyLinearVelocity)
		if LINEARIZE_ANGULAR then
			noteAngular(hrpNow.AssemblyAngularVelocity)
			linAngUntil = tick() + 1.0 -- a tumble outlasts a single TTL
		end
	end))
end

local function stateBlocksMovement(): boolean
	local hum = humanoid
	if not hum then
		return true
	end
	local st = hum:GetState()
	return st == Enum.HumanoidStateType.Ragdoll
		or st == Enum.HumanoidStateType.Physics
		or st == Enum.HumanoidStateType.FallingDown
		or st == Enum.HumanoidStateType.Seated
		or st == Enum.HumanoidStateType.Climbing
		or st == Enum.HumanoidStateType.Swimming
		or st == Enum.HumanoidStateType.PlatformStanding
		or st == Enum.HumanoidStateType.GettingUp
		or st == Enum.HumanoidStateType.Dead
end

-- specifically the "knocked / limp" states where you should NOT be moving under your own power
local function isKnockedState(): boolean
	local hum = humanoid
	if not hum then
		return false
	end
	local st = hum:GetState()
	return st == Enum.HumanoidStateType.Ragdoll
		or st == Enum.HumanoidStateType.Physics
		or st == Enum.HumanoidStateType.FallingDown
		or st == Enum.HumanoidStateType.PlatformStanding
		or st == Enum.HumanoidStateType.GettingUp
end

local function feetY(): number?
	local hrp = rootPart
	if not hrp then
		return nil
	end
	return hrp.Position.Y - footOffset
end

-- floor top Y, the floor part, and the floor normal beneath the character
local function groundInfo(): (number?, BasePart?, Vector3?)
	local hrp = rootPart
	if not hrp or not rayParams then
		return nil, nil, nil
	end
	local hit = workspace:Raycast(hrp.Position, Vector3.new(0, -(footOffset + 2.5), 0), rayParams)
	if not hit then
		return nil, nil, nil
	end
	return hit.Position.Y, hit.Instance :: BasePart, hit.Normal
end

-- genuine climbable step (near-vertical face + walkable top within STEP_HEIGHT)
local function detectStep(flat: Vector3, floorTopY: number, probe: number): number?
	local hrp = rootPart
	if not hrp or not rayParams then
		return nil
	end
	local faceOrigin = Vector3.new(hrp.Position.X, floorTopY + 0.2, hrp.Position.Z)
	local face = workspace:Raycast(faceOrigin, flat * probe, rayParams)
	if not face then
		return nil
	end
	if math.abs(face.Normal.Y) > 0.35 then
		return nil -- ramp, handled by slope glue
	end
	local ahead = face.Position + flat * 0.5
	local topOrigin = Vector3.new(ahead.X, floorTopY + STEP_HEIGHT + 1.0, ahead.Z)
	local top = workspace:Raycast(topOrigin, Vector3.new(0, -(STEP_HEIGHT + 1.5), 0), rayParams)
	if not top or top.Normal.Y < 0.7 then
		return nil
	end
	local h = top.Position.Y - floorTopY
	if h < 0.15 or h > STEP_HEIGHT then
		return nil
	end
	return top.Position.Y
end

local OURS = {
	LinearMove = true, LinearStep = true, LinearForce = true,
	LinearForceY = true, LinearSpin = true, LinearMoveAttachment = true,
}
local function suppressForeign()
	local hrp = rootPart
	if not hrp then
		return
	end
	for _, d in ipairs(hrp:GetChildren()) do
		if not OURS[d.Name] and (d:IsA("BodyMover") or d:IsA("Constraint")) then
			d:Destroy()
		end
	end
end

local function stopStep()
	stepping = false
	if stepLV and stepLV.Parent then
		stepLV.Enabled = false
		stepLV.LineVelocity = 0
	end
end

-- Diagnostics. With LINEARIZE_DEBUG on, nothing is converted -- every catch is just reported, so you
-- can learn what a given game actually does to you (which properties, from which script) before
-- trusting the transformer with it.
-- Queue a line for output. Kept cheap: the caller is often inside a metamethod hook running on every
-- property write in the game, so this must not print or touch the filesystem per event.
local function logEmit(line: string)
	local stamped = ("[%.2f] %s"):format(tick() % 100000, line)
	print("[LinearMovement] " .. stamped)
	if LOG_TO_FILE then
		table.insert(logBuf, stamped)
		if #logBuf > 400 then
			table.remove(logBuf, 1) -- bound memory if flushing is unavailable
		end
	end
end

-- Batch the buffer to disk. Called on a timer and on teardown, never per event.
local function flushLog()
	if not LOG_TO_FILE or #logBuf == 0 or not appendfile then
		return
	end
	local chunk = table.concat(logBuf, "\n") .. "\n"
	table.clear(logBuf)
	pcall(function()
		appendfile(LOG_FILE, chunk)
	end)
end

function linLog(channel: string, detail: string)
	if not LINEARIZE_DEBUG and not LINEARIZE_AUDIT then
		return
	end
	local now = tick()
	local last = logLast[channel]
	if last and now - last < LOG_THROTTLE then
		-- same channel again too soon: count it instead of printing, so a script that writes every
		-- frame shows up as a rate rather than thousands of identical lines
		logSkipped[channel] = (logSkipped[channel] or 0) + 1
		return
	end
	logLast[channel] = now
	local repeats = logSkipped[channel]
	logSkipped[channel] = 0
	-- resolving the calling script is expensive, so it only happens on lines we actually keep
	local src = "?"
	if getcallingscript then
		local ok, s = pcall(getcallingscript)
		if ok and s then
			local ok2, full = pcall(function() return s:GetFullName() end)
			src = (ok2 and full) or tostring(s)
		end
	end
	local suffix = (repeats and repeats > 0) and (" (+%d suppressed)"):format(repeats) or ""
	logEmit(("%s | %s | src=%s%s"):format(channel, detail, src, suffix))
end

-- Answers the "how does this game ragdoll me?" question: if a knock splits the character into several
-- assemblies (Motor6Ds swapped for BallSocketConstraints), one constraint on the HRP only drives the
-- torso and the limbs flop free. Logged on each knock while debugging.
function logAssemblies(tag: string)
	if not LINEARIZE_DEBUG then
		return
	end
	local hrp = rootPart
	local char = hrp and hrp.Parent
	if not hrp or not char then
		return
	end
	local roots, n = {}, 0
	for _, d in ipairs(char:GetDescendants()) do
		if d:IsA("BasePart") then
			local ok, root = pcall(function() return d.AssemblyRootPart end)
			if ok and root and not roots[root] then
				roots[root] = true
				n += 1
			end
		end
	end
	linLog("assembly/" .. tag, ("assemblies=%d hrpRoot=%s"):format(n, tostring(hrp.AssemblyRootPart)))
end

local function stopLinearize()
	linKind = nil
	linPosTarget = nil
	linNoY = false
	linVec = Vector3.zero
	linAngVec = Vector3.zero
	linAngUntil = 0
	if linLV and linLV.Parent then
		linLV.Enabled = false
		linLV.PlaneVelocity = Vector2.zero
	end
	if linYLV and linYLV.Parent then
		linYLV.Enabled = false
		linYLV.LineVelocity = 0
	end
	if linAV and linAV.Parent then
		linAV.Enabled = false
		linAV.AngularVelocity = Vector3.zero
	end
end

-- Record a caught intent. Classification happens here so every catch site stays dumb.
-- noY marks the very common "preserve my Y" idiom, e.g. Velocity = Vector3.new(x, Velocity.Y, z).
-- That is a horizontal command: the author is explicitly NOT changing vertical motion. Driving Y to
-- the sampled value would pin your fall speed and cancel gravity's acceleration.
function noteIntent(kind: string, vec: Vector3, noY: boolean?)
	local now = tick()
	linKind = kind
	linStart = now
	linNoY = noY == true
	if kind == "Damp" then
		-- bleed the momentum you ACTUALLY have; "stop" is a brake, not a position hold
		local hrp = rootPart
		linVec = hrp and hrp.AssemblyLinearVelocity or Vector3.zero
		linUntil = now + LINEARIZE_DAMP_TIME
	elseif kind == "Impulse" then
		linVec = vec
		linUntil = now + 1.5 -- decay curve ends it early in practice
	else
		linVec = vec
		linUntil = now + LINEARIZE_TTL
	end
end

function noteAngular(vec: Vector3)
	if not LINEARIZE_ANGULAR then
		return
	end
	linAngVec = vec
	linAngUntil = tick() + LINEARIZE_TTL
end

-- Drive the constraints from the current intent. Horizontal always goes through the plane
-- constraint; the Y constraint only engages when the intent is genuinely vertical (or when
-- LINEARIZE_GRAVITY is off), so you keep falling normally instead of hanging in the air.
local function applyLinear(vec: Vector3)
	local lin, linY = linLV, linYLV
	if lin and lin.Parent then
		lin.PlaneVelocity = Vector2.new(vec.X, vec.Z)
		lin.Enabled = true
	end
	if linY and linY.Parent then
		if linNoY then
			-- the source explicitly preserved Y; hands off the vertical axis
			linY.Enabled = false
			linY.LineVelocity = 0
		elseif not LINEARIZE_GRAVITY or math.abs(vec.Y) > 0.5 then
			linY.LineVelocity = vec.Y
			linY.Enabled = true
		else
			linY.Enabled = false
			linY.LineVelocity = 0
		end
	end
end

-- Leak detector. Compares your real velocity against what our constraints actually commanded and
-- reports the difference. Unlike the hooks, this needs no knowledge of HOW the velocity arrived, so
-- it is the only thing that can see channels we cannot hook -- the server, replication, or a script
-- that hooked the metamethod above us. Horizontal only, since gravity moves Y every frame by design.
local function auditVelocity()
	local hrp = rootPart
	if not LINEARIZE_AUDIT or not hrp then
		return
	end
	local now = tick()
	local vel = hrp.AssemblyLinearVelocity

	-- reconstruct what we asked for this frame across all our constraints
	local cx, cy, cz = 0, 0, 0
	if moveLV and moveLV.Parent and moveLV.Enabled then
		cx, cz = moveLV.PlaneVelocity.X, moveLV.PlaneVelocity.Y
	end
	if linLV and linLV.Parent and linLV.Enabled then
		cx, cz = linLV.PlaneVelocity.X, linLV.PlaneVelocity.Y
	end
	if stepLV and stepLV.Parent and stepLV.Enabled then
		cy = stepLV.LineVelocity
	end
	if linYLV and linYLV.Parent and linYLV.Enabled then
		cy = linYLV.LineVelocity
	end
	local cmd = Vector3.new(cx, cy, cz)

	local dActual = Vector3.new(vel.X - auditLastVel.X, 0, vel.Z - auditLastVel.Z).Magnitude
	local dCmd = Vector3.new(cmd.X - auditLastCmd.X, 0, cmd.Z - auditLastCmd.Z).Magnitude
	local unexplained = dActual - dCmd
	if unexplained > AUDIT_TOLERANCE and now - auditLastReport > AUDIT_COOLDOWN then
		auditLastReport = now
		local hum = humanoid
		local driving = (linLV and linLV.Enabled) or (linYLV and linYLV.Enabled)
		logEmit(("AUDIT unexplained=%.1f actual=%.1f cmd=%.1f state=%s floor=%s moveDir=%.2f linearDriving=%s")
			:format(
				unexplained,
				Vector3.new(vel.X, 0, vel.Z).Magnitude,
				Vector3.new(cmd.X, 0, cmd.Z).Magnitude,
				hum and tostring(hum:GetState()) or "?",
				hum and tostring(hum.FloorMaterial) or "?",
				hum and hum.MoveDirection.Magnitude or 0,
				tostring(driving and true or false)
			))
	end
	auditLastVel = vel
	auditLastCmd = cmd
end

-- Replay whatever foreign motion the hook caught, through the constraints instead of a raw write.
-- Priority: a caught teleport outranks a velocity intent, since it is a destination, not a rate.
local function updateLinearize()
	local hrp = rootPart
	local hum = humanoid
	if not LINEARIZE_FOREIGN or LINEARIZE_DEBUG or not linLV or not linLV.Parent or not hrp then
		stopLinearize()
		return
	end
	if hum and hum.Health <= 0 then
		stopLinearize()
		return
	end
	local now = tick()

	-- spin decays on its own clock so a tumble can outlive the travel intent
	if linAV and linAV.Parent then
		if LINEARIZE_ANGULAR and now < linAngUntil then
			linAV.AngularVelocity = linAngVec
			linAV.Enabled = true
		else
			linAV.Enabled = false
			linAV.AngularVelocity = Vector3.zero
		end
	end

	if linPosTarget then
		local delta = linPosTarget - hrp.Position
		if delta.Magnitude <= LINEARIZE_REACH then
			stopLinearize() -- arrived: hand back to normal physics
		else
			-- speed scales with distance and eases as you arrive, capped at LINEARIZE_SPEED
			local v = delta.Unit * math.min(delta.Magnitude * 8, LINEARIZE_SPEED)
			-- a teleport IS a vertical command, so drive Y regardless of the gravity preference
			if linLV and linLV.Parent then
				linLV.PlaneVelocity = Vector2.new(v.X, v.Z)
				linLV.Enabled = true
			end
			if linYLV and linYLV.Parent then
				linYLV.LineVelocity = v.Y
				linYLV.Enabled = true
			end
		end
		return
	end

	if not linKind or now >= linUntil then
		stopLinearize()
		return
	end

	local elapsed = now - linStart
	if linKind == "Damp" then
		-- ramp the captured momentum down to zero across LINEARIZE_DAMP_TIME, then release
		local t = math.clamp(1 - elapsed / LINEARIZE_DAMP_TIME, 0, 1)
		applyLinear(linVec * t)
	elseif linKind == "Impulse" then
		-- exponential settle so a knock lands naturally instead of cutting off at a TTL
		local decay = math.exp(-LINEARIZE_DECAY * elapsed)
		if decay < 0.05 then
			stopLinearize()
		else
			applyLinear(linVec * decay)
		end
	else -- "Set"
		applyLinear(linVec)
	end
end

local function disableAll()
	if moveLV and moveLV.Parent then
		moveLV.Enabled = false
		moveLV.PlaneVelocity = Vector2.zero
	end
	airFloorVX, airFloorVZ = 0, 0
	stopStep()
end

table.insert(connections, UserInputService.JumpRequest:Connect(function()
	jumpSuppressUntil = tick() + 0.25
	stopStep()
end))

table.insert(connections, RunService.Heartbeat:Connect(function(dt)
	local move = moveLV
	local hum = humanoid
	local hrp = rootPart
	if not move or not move.Parent or not hum or not hrp then
		return
	end

	if SUPPRESS_OTHER then
		suppressForeign()
	end
	if MASK_WALKSPEED > 0 and hum.WalkSpeed ~= MASK_WALKSPEED then
		hum.WalkSpeed = MASK_WALKSPEED
	end

	-- Track the fastest speed the game has given you this life. A drop below it means the game
	-- deliberately slowed you -- carrying something, a debuff, a zone. Note this is read BEFORE any
	-- standdown so the baseline keeps updating while slowed.
	if hum.WalkSpeed > baseWalkSpeed then
		baseWalkSpeed = hum.WalkSpeed
	end
	local slowed = CARRY_SAFE and baseWalkSpeed > 0 and hum.WalkSpeed < baseWalkSpeed - CARRY_SLOW_MARGIN

	-- Diagnostics run before any standdown, so a carry window is still observable -- that window is
	-- usually the one worth reading.
	auditVelocity()
	if LOG_TO_FILE and tick() >= logNextFlush then
		logNextFlush = tick() + LOG_FLUSH_EVERY
		flushLog()
	end

	if slowed and not LINEAR_PRIMARY then
		-- Booster mode: there is nothing to do while slowed except get out of the way, since the only
		-- thing the constraint would add is speed the game just took away.
		linLog("carry", ("slowed: ws=%.1f base=%.1f -> standing down"):format(hum.WalkSpeed, baseWalkSpeed))
		disableAll()
		stopLinearize()
		return
	end

	-- the linearizer runs on its own, independent of the speed booster below
	updateLinearize()

	if hum.Health <= 0 or stateBlocksMovement() then
		disableAll()
		return
	end

	local gameWS = hum.WalkSpeed
	local rooted = gameWS <= 0.01
	local effectiveSpeed = (SPEED_FIXED > 0) and SPEED_FIXED or (gameWS * SPEED_MULT)
	-- While the game has you slowed, never exceed the speed it just handed you. In primary mode we
	-- keep driving (the constraint stays the mover) but at exactly the game's speed, so the mechanism
	-- is still linear and the speed is still whatever the game says it should be.
	if slowed then
		effectiveSpeed = gameWS
	end
	local boosting = (not rooted) and effectiveSpeed > gameWS + 0.1
	-- primary mode drives every step; booster mode only drives the part above the game's speed
	local driving = boosting or (LINEAR_PRIMARY and not rooted)

	if not driving then
		-- vanilla native movement handles everything (speed, slopes, stairs, anim, rooting)
		disableAll()
		return
	end

	local floorTopY, floorPart, floorNormal = groundInfo()
	local fy = feetY()
	local grounded = floorTopY ~= nil and fy ~= nil and (fy - floorTopY) <= GROUND_TOLERANCE

	local dir = hum.MoveDirection
	local moving = dir.Magnitude > 0.01
	local flat = moving and Vector3.new(dir.X, 0, dir.Z).Unit or Vector3.zero
	local liftCap = math.max(STEP_UP_SPEED, effectiveSpeed)

	-- vertical control via stepLV: step-assist (climb ledges) + slope-glue (hug ramps).
	-- an in-progress step is NOT gated on grounded (it lifts you off the ground by design).
	local now = tick()
	if now < jumpSuppressUntil or not moving then
		stopStep()
	elseif stepping then
		local nowFy = feetY()
		if nowFy == nil or nowFy >= stepTargetY - 0.1 or now > stepDeadline then
			stopStep()
		elseif stepLV and stepLV.Parent then
			stepLV.LineVelocity = math.clamp((stepTargetY - nowFy) * 20, 4, liftCap)
			stepLV.Enabled = true
		end
	elseif grounded then
		local handled = false
		if STEP_ASSIST and floorTopY and fy then
			local topY = detectStep(flat, floorTopY, STEP_PROBE + effectiveSpeed * 0.04)
			if topY and (topY - fy) > 0.25 and stepLV and stepLV.Parent then
				stepping = true
				stepTargetY = topY
				stepDeadline = now + 0.6
				stepLV.LineVelocity = math.clamp((topY - fy) * 20, 4, liftCap)
				stepLV.Enabled = true
				handled = true
			end
		end
		if not handled and stepLV and stepLV.Parent then
			if SLOPE_GLUE and floorNormal and floorNormal.Y > 0.1 and floorNormal.Y < 0.999 then
				-- hug the slope: drive Y so velocity stays tangent to the surface (up or down)
				local vhx = dir.X * effectiveSpeed
				local vhz = dir.Z * effectiveSpeed
				local vy = -(vhx * floorNormal.X + vhz * floorNormal.Z) / floorNormal.Y
				local cap = effectiveSpeed * 3.5 + 5
				vy = math.clamp(vy, -cap, cap)
				stepLV.LineVelocity = vy
				stepLV.Enabled = true
				lastGlueTime = now
				lastGlueVy = vy
			elseif lastGlueVy > 0 and now - lastGlueTime < 0.2 then
				-- just crested from an upslope onto flat: bleed the residual upward velocity so you don't launch
				stepLV.LineVelocity = 0
				stepLV.Enabled = true
			else
				stopStep()
			end
		end
	else
		stopStep()
	end

	-- inherit the floor's motion so moving platforms carry you; decays after you leave one
	local fvx, fvz = 0, 0
	if grounded and floorPart and floorTopY then
		local ok, v = pcall(function()
			return floorPart:GetVelocityAtPosition(Vector3.new(hrp.Position.X, floorTopY, hrp.Position.Z))
		end)
		if ok and v then
			fvx = math.abs(v.X) > FLOOR_VEL_DEADZONE and v.X or 0
			fvz = math.abs(v.Z) > FLOOR_VEL_DEADZONE and v.Z or 0
		end
		airFloorVX, airFloorVZ = fvx, fvz
	else
		local decay = math.max(0, 1 - dt * 3)
		airFloorVX *= decay
		airFloorVZ *= decay
	end
	local baseVX = grounded and fvx or airFloorVX
	local baseVZ = grounded and fvz or airFloorVZ

	-- let external knockback/impulses through: if we're moving much faster horizontally than we
	-- asked for, something is shoving us, so back off the constraint briefly and let it play out.
	local av = hrp.AssemblyLinearVelocity
	local hspeed = Vector3.new(av.X, 0, av.Z).Magnitude
	local expected = effectiveSpeed + Vector2.new(baseVX, baseVZ).Magnitude
	if hspeed > expected + KNOCK_THRESHOLD then
		knockUntil = now + 0.3
	end

	if now < knockUntil then
		move.Enabled = false
		move.PlaneVelocity = Vector2.zero
	elseif moving and (grounded or AIR_CONTROL or stepping) then
		move.PlaneVelocity = Vector2.new(dir.X * effectiveSpeed + baseVX, dir.Z * effectiveSpeed + baseVZ)
		move.Enabled = true
	else
		-- idle (or airborne with no air control): let native movement hold/carry you
		move.Enabled = false
		move.PlaneVelocity = Vector2.zero
	end
end))

local function cleanup()
	alive = false
	for _, c in ipairs(connections) do
		pcall(function() c:Disconnect() end)
	end
	table.clear(connections)
	stopStep()
	stopLinearize()
	logEmit("session end (cleanup)")
	flushLog()
	teardown()
	_G.__LinearMovementCleanup = nil
end
_G.__LinearMovementCleanup = cleanup

if (BLOCK_FOREIGN_VELOCITY or SUPPRESS_DURING_RAGDOLL or LINEARIZE_FOREIGN or LINEARIZE_DEBUG)
	and hookmetamethod and newcclosure then
	pcall(function()
		local realNewindex
		realNewindex = hookmetamethod(game, "__newindex", newcclosure(function(self, key, value)
			-- A velocity write to ANY part of the assembly moves all of you, so match the whole rig,
			-- not just the HRP. ourParts is a precomputed set because this runs on every property
			-- write in the game and must not walk the tree.
			if self == rootPart or (LINEARIZE_WHOLE_RIG and ourParts[self]) then
				if (key == "Velocity" or key == "AssemblyLinearVelocity") and typeof(value) == "Vector3" then
					local knocked = isKnockedState()
					linLog("velocity", ("%s=%s knocked=%s"):format(key, tostring(value), tostring(knocked)))
					-- Transformer FIRST: a caught write becomes an intent and is replayed through the
					-- constraints. This deliberately runs before the ragdoll-suppress branch below, so
					-- getting knocked comes out linear instead of the write landing as raw velocity.
					-- A zero write is a Damp (brake), not a Set -- otherwise a script that zeroes every
					-- frame would pin you in mid-air.
					if LINEARIZE_FOREIGN and not LINEARIZE_DEBUG and (not knocked or LINEARIZE_RAGDOLL) then
						-- Reading self[key] goes through __index, which we do not hook, so this cannot
						-- recurse. If the new Y matches the current Y the writer is preserving it.
						local okCur, cur = pcall(function() return self[key] end)
						local noY = okCur and typeof(cur) == "Vector3" and math.abs(value.Y - cur.Y) < 0.5
						noteIntent(value.Magnitude < 0.05 and "Damp" or "Set", value, noY)
						return
					end
					-- Fallback when the transformer is off (or is set to stay out of knocks): swallow
					-- foreign velocity writes while limp so the ragdoll plays out and the server doesn't
					-- kick you for moving. Walking/jumps aren't affected.
					if SUPPRESS_DURING_RAGDOLL and knocked then
						return
					end
					-- Optional: also swallow velocity writes on our HRP WHILE we're actively walking, so
					-- another script's walk mover is replaced by ours (its TP/auto-grab still work).
					if BLOCK_FOREIGN_VELOCITY then
						local hum = humanoid
						if hum and hum.MoveDirection.Magnitude > 0.01 then
							-- only swallow horizontal (walk) overrides; let vertical writes (jumps) through
							local cur = self[key]
							if math.abs(value.Y - cur.Y) < 1 then
								return
							end
						end
					end
				elseif (key == "RotVelocity" or key == "AssemblyAngularVelocity") and typeof(value) == "Vector3" then
					linLog("angular", ("%s=%s"):format(key, tostring(value)))
					if LINEARIZE_FOREIGN and LINEARIZE_ANGULAR and not LINEARIZE_DEBUG then
						noteAngular(value)
						return
					end
				-- Teleports are matched on the HRP ONLY. Limb CFrames are written constantly by
				-- animations and welds; swallowing those would tear the character apart.
				elseif self == rootPart and key == "CFrame" and typeof(value) == "CFrame" then
					linLog("teleport", "CFrame=" .. tostring(value.Position))
					-- Transformer: catch a teleport and glide to its target through the constraints
					-- instead of letting the CFrame snap you there.
					if LINEARIZE_FOREIGN and LINEARIZE_TELEPORTS and not LINEARIZE_DEBUG then
						linPosTarget = value.Position
						return
					end
				elseif self == rootPart and key == "Position" and typeof(value) == "Vector3" then
					linLog("teleport", "Position=" .. tostring(value))
					if LINEARIZE_FOREIGN and LINEARIZE_TELEPORTS and not LINEARIZE_DEBUG then
						linPosTarget = value
						return
					end
				end
			end
			return realNewindex(self, key, value)
		end))
	end)
end

-- __namecall: the other half of the coverage. Nothing below writes a property, so none of it is
-- visible to __newindex -- a script using :ApplyImpulse() or :PivotTo() would move you raw no matter
-- what the property hook did. Impulses are converted to the velocity change they represent
-- (impulse / mass) so they can be reproduced as a decaying Impulse intent.
-- Only install this if it will actually do something. A __namecall hook runs on EVERY method call in
-- the game, so installing it while conversion is off buys nothing and costs a pcall plus a lookup on
-- every call -- enough to matter in a game with heavy remote traffic.
if (LINEARIZE_FOREIGN and LINEARIZE_METHODS or LINEARIZE_DEBUG)
	and hookmetamethod and newcclosure and getnamecallmethod then
	pcall(function()
		local realNamecall
		realNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
			local okM, method = pcall(getnamecallmethod)
			if not okM or not method then
				return realNamecall(self, ...)
			end
			local hrp = rootPart
			local isPart = ourParts[self] ~= nil
			local isRig = isPart or (hrp and self == hrp.Parent)
			if not isRig then
				return realNamecall(self, ...)
			end
			local convert = LINEARIZE_FOREIGN and LINEARIZE_METHODS and not LINEARIZE_DEBUG

			if method == "ApplyImpulse" or method == "ApplyImpulseAtPosition" then
				local impulse = (select(1, ...))
				if typeof(impulse) == "Vector3" and hrp then
					linLog("namecall", ("%s=%s"):format(method, tostring(impulse)))
					if convert then
						local mass = 1
						local okMass, m = pcall(function() return hrp.AssemblyMass end)
						if okMass and m and m > 0 then
							mass = m
						end
						-- an impulse is a momentum change: convert to the velocity it would produce
						noteIntent("Impulse", hrp.AssemblyLinearVelocity + impulse / mass)
						return
					end
				end
			elseif method == "ApplyAngularImpulse" then
				local impulse = (select(1, ...))
				if typeof(impulse) == "Vector3" then
					linLog("namecall", "ApplyAngularImpulse=" .. tostring(impulse))
					if convert and LINEARIZE_ANGULAR then
						noteAngular(impulse)
						return
					end
				end
			elseif method == "PivotTo" and typeof((select(1, ...))) == "CFrame" then
				local cf = (select(1, ...))
				linLog("namecall", "PivotTo=" .. tostring(cf.Position))
				if convert and LINEARIZE_TELEPORTS then
					linPosTarget = cf.Position
					return
				end
			elseif method == "MoveTo" and self ~= humanoid and typeof((select(1, ...))) == "Vector3" then
				-- Model:MoveTo is a teleport. Humanoid:MoveTo is a walk request, so it is left alone.
				local pos = (select(1, ...))
				linLog("namecall", "MoveTo=" .. tostring(pos))
				if convert and LINEARIZE_TELEPORTS then
					linPosTarget = pos
					return
				end
			elseif method == "TranslateBy" and typeof((select(1, ...))) == "Vector3" and hrp then
				local off = (select(1, ...))
				linLog("namecall", "TranslateBy=" .. tostring(off))
				if convert and LINEARIZE_TELEPORTS then
					linPosTarget = hrp.Position + off
					return
				end
			end
			return realNamecall(self, ...)
		end))
	end)
end
